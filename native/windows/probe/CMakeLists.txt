cmake_minimum_required(VERSION 3.25)
project(bunaway_windows_probe LANGUAGES CXX)
add_executable(bunaway-probe host.cpp)
set_property(TARGET bunaway-probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_features(bunaway-probe PRIVATE cxx_std_20)
target_compile_definitions(bunaway-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(bunaway-probe PRIVATE ../vendor)
target_compile_options(bunaway-probe PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(bunaway-probe PRIVATE bcrypt)
